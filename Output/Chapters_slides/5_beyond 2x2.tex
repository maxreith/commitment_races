\section{Beyond $2\times2$}
\begin{frame}{A $3\times3$ game – commitment to dominated actions}
\centering
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{c|ccc}
      & $A_1$ & $A_2$ & $A_3$ \\\hline
$A_1$ & $1,1$ & $3,0$   & $\epsilon,0$ \\
$A_2$ & $0,\epsilon$ & $2,1$ & $0,0$ \\
$A_3$ & $0,3$ & $0,0$   & $1,2$
\end{tabular}

\bigskip
\begin{itemize}
  \item Unique base-game NE: \((A_1,A_1)\).
  \item Yet in the commitment game each player may \emph{credibly commit}
        to the \emph{dominated} actions \(A_2\) or \(A_3\).  
  \item Interior mixed equilibrium: commit to \(A_2\)/\(A_3\) w.p.\(\tfrac12\); expected payoff \(=1\).
\end{itemize}
\end{frame}
\begin{frame}{Modified $3\times3$ – commitment turns catastrophic}
\centering
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{c|ccc}
      & $A_1$ & $A_2$ & $A_3$ \\\hline
$A_1$ & $1,1$ & $3,0$ & $\epsilon-\textcolor{red}{10^{9}},\,0$ \\
$A_2$ & $0,\,\epsilon-\textcolor{red}{10^{9}}$
      & $2,\,1-\textcolor{red}{10^{9}}$
      & $-\textcolor{red}{10^{9}},\,-\textcolor{red}{10^{9}}$ \\
$A_3$ & $0,3$ & $0,0$
      & $1-\textcolor{red}{10^{9}}, 2$
\end{tabular}

\bigskip
\begin{itemize}
  \item Base-game NE still \((A_1,A_1)\) with payoff \(1\).
  \item Commitment game’s interior mixed equilibrium now yields
        \(\displaystyle 1-\frac{10^{9}}{2}\) each → enormous welfare loss
\end{itemize}
\end{frame}
