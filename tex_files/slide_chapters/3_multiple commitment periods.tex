\begin{frame}{Proposition 2}
\begin{block}{Proposition}
As \(T \to \infty\), the expected payoff in the unique mixed subgame-perfect equilibrium for both players
\begin{enumerate}
    \item decreases in the Game of Chicken and converges to \(c\);
    \item increases in the Battle of Sexes and converges to \(\min\{b,c\}\).
\end{enumerate}
\end{block}
